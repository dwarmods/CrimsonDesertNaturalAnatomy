cmake_minimum_required(VERSION 3.20)
project(NeutralAnatomyRuntime LANGUAGES C CXX ASM_MASM)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(neutral_motion STATIC src/motion.cpp)
target_include_directories(neutral_motion PUBLIC src)
add_executable(neutral_motion_tests tests/motion_tests.cpp)
target_link_libraries(neutral_motion_tests PRIVATE neutral_motion)
enable_testing()
add_test(NAME neutral_motion_constraints COMMAND neutral_motion_tests)
add_subdirectory(vendor/minhook)
add_library(NeutralRuntimeProbe SHARED src/probe.cpp src/probe_hook.asm)
target_include_directories(NeutralRuntimeProbe PRIVATE vendor/minhook/include)
target_compile_definitions(NeutralRuntimeProbe PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
target_link_libraries(NeutralRuntimeProbe PRIVATE minhook user32)
set_target_properties(NeutralRuntimeProbe PROPERTIES PREFIX "" SUFFIX ".asi")
add_executable(neutral_probe_abi_tests tests/probe_abi_tests.cpp src/probe_hook.asm)
target_include_directories(neutral_probe_abi_tests PRIVATE src vendor/minhook/include)
target_compile_definitions(neutral_probe_abi_tests PRIVATE PROBE_ABI_TEST WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(neutral_probe_abi_tests PRIVATE minhook)
add_test(NAME upload_probe_argument_preservation COMMAND neutral_probe_abi_tests)
